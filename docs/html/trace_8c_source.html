147void vtrace(
148 TraceCategory cat, int line, const char *filename, TraceType type,
149 const char *fmt, va_list vargs
150) {
151 if ((g_cat_filter & cat) || (g_type_filter & type)) {
152 return;
153 }
154
155 if (!g_stream) {
156 g_stream = stdout;
157 }
158
159 unsigned cat_idx = cat_to_idx(cat);
160 unsigned type_idx = type_to_idx(type);
161 const char *cat_str = g_cat_str[cat_idx];
162 const char *type_str = g_type_str[type_idx];
163
164#if _WIN32
165 if (_isatty(_fileno(g_stream))) {
166#elif __unix__
167 if (isatty(fileno(g_stream))) {
168#else
169 if (0) {
170#endif
171 fprintf(
172 g_stream, "\033[2m%s%sTRACE %s%s:%d %s: ", g_cat_color_mod[cat_idx],
173 g_type_color_mod[type_idx], cat_str, filename, line, type_str
174 );
175 vfprintf(g_stream, fmt, vargs);
176 fputs("\033[0m\n", g_stream);
177 } else {
178 fprintf(
179 g_stream, "TRACE [%s] %s:%d %s: ", cat_str, filename, line, type_str
180 );
181 vfprintf(g_stream, fmt, vargs);
182 fputc('\n', g_stream);
183 }
184}
