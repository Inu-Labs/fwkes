40static void *load_file(Fs *fs, const char *path, unsigned *size) {
41 File file;
42
43 if (fs_open(fs, &file, path, "rb") != FS_ERR_OK) {
44 return NULL;
45 }
46
47 size_t size_tmp;
48 if (file_size(&file, &size_tmp) != FS_ERR_OK) {
49 file_close(&file);
50 return NULL;
51 }
52
53 *size = (unsigned) size_tmp;
54
55 void *buf = disk_malloc(*size);
56
57 if (!buf) {
58 file_close(&file);
59
60 return NULL;
61 }
62
63 if (file_read(&file, buf, *size) != FS_ERR_OK) {
64 disk_free(buf);
65 buf = NULL;
66 }
67
68 file_close(&file);
69
70 return buf;
71}

73FORCE_INLINE MapperId mapper(const Disk *self) {
74 switch (self->fmt) {
75 case DISK_ERR:
76 return 0;
77 case DISK_INES:
78 return ((self->ines.flags6 & INES_FLAGS6_MAPPER_LN) >> 4) |
79 (self->ines.flags7 & INES_FLAGS7_MAPPER_HN);
80 }
81
82 return 0;
83}

107bool disk_ines_parse(DiskINes *self, const uint8_t *rom) {
108 if (memcmp(rom, "NES\x1a", 4) != 0) {
109 log_msg(
110 LOG_WARN,
111 "bad iNES/NES 2.0 signature: 0x%02x 0x%02x 0x%02x 0x%02x (must be "
112 "0x%02x 0x%02x 0x%02x 0x%02x)",
113 rom[0], rom[1], rom[2], rom[3], 'N', 'E', 'S', '\x1a'
114 );
115
116 return false;
117 }
118
119 self->prg_rom_size = rom[4];
120 self->chr_rom_size = rom[5];
121 self->flags6 = rom[6];
122 self->flags7 = rom[7];
123 self->flags8 = rom[8];
124 self->flags9 = rom[9];
125 self->flags10 = rom[10];
126
127 return true;
128}

130static bool parse(Disk *self) {
131 if (disk_ines_parse(&self->ines, self->rom)) {
132 self->fmt = DISK_INES;
133 } else {
134 disk_unload(self);
135
136 return false;
137 }
138
139 self->mapper = mapper(self);
140 size_t offset = 0x10;
141
142 if (self->ines.flags6 & 0x04) {
143 offset += 512;
144 }
145
146 self->prg_size = prg_size(self);
147 self->chr_size = chr_size(self);
148
149 self->prg = &self->rom[offset];
150 self->chr = &self->rom[offset + self->prg_size];
151 self->prg_ram = disk_calloc(1, PRG_RAM_SIZE);
152 self->mirroring = (self->ines.flags6 & INES_FLAGS6_NAMETABLE_MIRRORING)
153 ? MIRRORING_VERTICAL
154 : MIRRORING_HORIZONTAL;
155
156 if (!self->prg_ram) {
157 disk_unload(self);
158
159 return false;
160 }
161
162 switch (self->mapper) {
163 case 0:
164 self->map0.prg_c000_addr =
165 self->prg_size == PRG_ROM_UNIT * 1 ? 0 : 0x4000;
166 self->map0.prg_8000 = &self->prg[0];
167 self->map0.prg_c000 = &self->prg[self->map0.prg_c000_addr];
168
169 self->mapper_write = mapper0_write;
170 self->mapper_read = mapper0_read;
171 self->mapper_ppu_read = NULL;
172 self->mapper_ppu_write = NULL;
173 self->mapper_hsync = NULL;
174
175 break;
176 case 2:
177 if (self->chr_size == 0) {
178 self->chr_is_ram = true;
179 self->chr_size = 0x2000;
180 self->chr = disk_calloc(1, 0x2000);
181 }
182
183 self->map2.prg_bank_count = (uint8_t) (self->prg_size >> 14);
184 self->map2.prg_bank = 0;
185 self->map2.prg_8000 = &self->prg[0];
186 self->map2.prg_c000 =
187 &self->prg[(self->map2.prg_bank_count - 1) * 0x4000];
188
189 self->mapper_write = mapper2_write;
190 self->mapper_read = mapper2_read;
191 self->mapper_ppu_read = NULL;
192 self->mapper_ppu_write = NULL;
193 self->mapper_hsync = NULL;
194
195 break;
196
197 case 4:
198 if (self->chr_size == 0) {
199 self->chr_is_ram = true;
200 self->chr_size = 0x2000;
201 self->chr = disk_calloc(1, 0x2000);
202 }
203
204 Mapper4 *m = &self->map4;
205
206 memset(m, 0, sizeof(Mapper4));
207
208 m->prg_bank_count = self->prg_size / 0x2000;
209 if (m->prg_bank_count == 0) {
210 m->prg_bank_count = 1;
211 }
212
213 m->irq_cnt = 0;
214 m->irq_enabled = false;
215 m->irq_reload_pending = false;
216 m->irq_latch = 0;
217
218 mapper4_update_prg(self);
219 mapper4_update_chr(self);
220
221 self->mapper_write = mapper4_write;
222 self->mapper_read = mapper4_read;
223 self->mapper_ppu_read = mapper4_ppu_read;
224 self->mapper_ppu_write = mapper4_ppu_write;
225 self->mapper_hsync = mapper4_hsync;
226
227 break;
228 default:
229 log_msg(LOG_ERROR, "unknown mapper %u\n", self->mapper);
230 disk_unload(self);
231
232 return false;
233 }
234
235 return true;
236}

238bool disk_load(Disk *self, Bus *bus, Fs *fs, const char *path) {
239 memset(self, 0, sizeof(*self));
240 self->rom = load_file(fs, path, &self->size);
241 self->bus = bus;
242
243 if (!self->rom) {
244 return false;
245 }
246
247 return parse(self);
248}

250bool disk_load_mem(Disk *self, Bus *bus, const uint8_t *data, unsigned size) {
251 memset(self, 0, sizeof(*self));
252 self->rom = disk_malloc(size);
253 self->bus = bus;
254
255 if (!self->rom) {
256 return false;
257 }
258
259 memcpy(self->rom, data, size);
260
261 return parse(self);
262}

264void disk_unload(Disk *self) {
265 disk_free(self->prg_ram);
266 disk_free(self->rom);
267
268 if (self->chr_is_ram) {
269 disk_free(self->chr);
270 }
271
272 memset(self, 0, sizeof(*self));
