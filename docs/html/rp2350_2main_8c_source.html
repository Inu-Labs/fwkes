86static void __not_in_flash_func(prepare_dvi_scanline)(void) {
87 static uint16_t scanline[VIDEO_WIDTH] = {0};
88 static unsigned curr_scanline;
89 static uint16_t __attribute__((aligned(4))) colors_lut[32];
90
91 if (g_ppu->colors_lut_dirty) {
92 memcpy(colors_lut, g_ppu->colors_lut, 32 * sizeof(PpuPixel));
93 g_ppu->colors_lut_dirty = false;
94 }
95
96 uint8_t *fb_scanline = g_framebuffer[curr_scanline];
97 uint16_t *dvi_scanline_ptr = scanline + LEFT_PADDING;
98
99 for (unsigned i = 0; i < PPU_WIDTH; ++i) {
100 *dvi_scanline_ptr++ = colors_lut[*fb_scanline++ & 0x1f];
101 }
102
103 if (++curr_scanline == 240) {
104 curr_scanline = 0;
105 }
106
107 const uint16_t *scanbuf = scanline;
108 queue_add_blocking_u32(&g_dvi.q_colour_valid, &scanbuf);
109 while (queue_try_remove_u32(&g_dvi.q_colour_free, &scanbuf))
110 ;
111}

151static void run_frame(Emulator *self) {
152 Cpu *cpu = &self->bus.cpu;
153 Ppu *ppu = &self->bus.ppu;
154 Apu *apu = &self->bus.apu;
155
156 while (!ppu->frame_done) {
157 /* CPU will run until some significant event (vblank/nmi or end of
158 * frame)
159 */
160
161 CycleCounter curr_ppu_abs_dot =
162 ppu->scanline * DOTS_PER_SCANLINE + ppu->dot;
163 CycleCounter target_ppu_abs_dot;
164
165 if (self->bus.disk.mapper_hsync && ppu->scanline < 240) {
166 if (ppu->dot < 260) {
167 /* Target the current scanline's IRQ dot */
168 target_ppu_abs_dot = ppu->scanline * DOTS_PER_SCANLINE + 260;
169 } else {
170 /* We passed it, target the NEXT scanline's IRQ dot */
171 target_ppu_abs_dot =
172 (ppu->scanline + 1) * DOTS_PER_SCANLINE + 260;
173 }
174 } else if (ppu->scanline < 241) {
175 /* next event is vblank */
176 target_ppu_abs_dot = 241 * DOTS_PER_SCANLINE + 1;
177 } else {
178 /* next event is end of frame */
179 target_ppu_abs_dot = 262 * DOTS_PER_SCANLINE;
180 }
181
182 CycleCounter cycles_to_event =
183 (target_ppu_abs_dot - curr_ppu_abs_dot) / 3;
184 CycleCounter event_deadline = cpu->cycles + cycles_to_event;
185
186 CycleCounter cycles_to_run = cpu->cycles + CYCLES_PER_SCANLINE;
187
188 /* significant event has higher priority */
189 if (event_deadline < cycles_to_run) {
190 cycles_to_run = event_deadline;
191 }
192
193 /* prevent infinite loop */
194 if (cycles_to_run <= cpu->cycles) {
195 cycles_to_run = cpu->cycles + 1;
196 }
197
198 cpu_run_until(cpu, cycles_to_run);
199 ppu_run_until(ppu, cpu->cycles * 3);
200 apu_run_until(apu, cpu->cycles);
201 }
202
203 ppu->frame_done = false;
204
205 /* The ultimate fix to prevent overflow of 32-bit cycle counter */
206 if (cpu->cycles >= 1000000000) {
207 cpu->cycles -= 1000000000;
208 ppu->cycles -= 1000000000u * 3;
209 apu->cycles -= 1000000000;
210 }
211}

325 printf("Successfully initialized fs\n");
326 }
327

333

335
336 g_emu.bus.reset_cb = reset_callback;
337 g_emu.bus.user_data = &g_emu;
338 g_emu.bus.ppu.scanline_cb = prepare_ppu_scanline;
339 g_emu.bus.ppu.user_data = NULL;
340 g_emu.bus.apu.sample_cb = sample_callback;
341 g_emu.bus.apu.user_data = &g_emu;
342 g_emu.bus.ppu.colors = g_colors_rgb565;
343
344 ppu_init_pixel_luts(&g_emu.bus.ppu);
345
346 printf("Initializing APU DMA channel...\n");
347
348 audio_init();
349

352 printf("Cannot load BIOS\n");

356 }

358 printf("BIOS successfully loaded\n");

365failure:
