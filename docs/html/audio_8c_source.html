40FORCE_INLINE void init_pwm(void) {
41 gpio_set_function(AUDIO_MONO_PIN, GPIO_FUNC_PWM);
42
43 g_pwm_slice = pwm_gpio_to_slice_num(AUDIO_MONO_PIN);
44
45 unsigned sys_clk = clock_get_hz(clk_sys);
46 unsigned pwm_freq = 250000;
47 g_pwm_wrap = (sys_clk / pwm_freq - 1) & 0xffff;
48
49 pwm_set_wrap(g_pwm_slice, g_pwm_wrap);
50 pwm_set_clkdiv(g_pwm_slice, 1.f);
51 pwm_set_enabled(g_pwm_slice, true);
52}

54void audio_init(void) {
55 for (int i = 0; i < AUDIO_BUFFER_SIZE; ++i) {
56 g_bufs[0][i] = PWM_VALUE(0);
57 g_bufs[1][i] = PWM_VALUE(0);
58 }
59
60 for (int i = 0; i < RING_SIZE; ++i) {
61 g_reload_addrs[i] = (uint16_t *) ((uintptr_t) g_bufs[i] | 0x20000000);
62 }
63
64 init_pwm();
65
66 g_dma_timer = (unsigned) dma_claim_unused_timer(true);
67 /* DMA timer use formula sys_clk * (numerator / denominator).
68 * But we can't directly pass sys_clk, since it can't fit into uint16_t (300
69 * MHz). The workaround is to simplify the fraction (31.25kHz/300MHz =
70 * 1/9600). dma_timer_set_fraction(g_dma_timer, SAMPLE_RATE, sys_clk); */
71 dma_timer_set_fraction(g_dma_timer, 1, 9600);
72
73 g_audio_chan = (unsigned) dma_claim_unused_channel(true);
74 g_ctrl_chan = (unsigned) dma_claim_unused_channel(true);
75
76 dma_channel_config audio_cfg = dma_channel_get_default_config(g_audio_chan);
77 dma_channel_config ctrl_cfg = dma_channel_get_default_config(g_ctrl_chan);
78
79 channel_config_set_transfer_data_size(&audio_cfg, DMA_SIZE_16);
80 channel_config_set_read_increment(&audio_cfg, true);
81 channel_config_set_write_increment(&audio_cfg, false);
82 channel_config_set_dreq(&audio_cfg, dma_get_timer_dreq(g_dma_timer));
83 channel_config_set_chain_to(&audio_cfg, g_ctrl_chan);
84 channel_config_set_high_priority(&audio_cfg, true);
85
86 dma_channel_configure(
87 g_audio_chan, &audio_cfg, (void *) &pwm_hw->slice[g_pwm_slice].cc,
88 g_bufs[0], AUDIO_BUFFER_SIZE, false
89 );
90
91 channel_config_set_transfer_data_size(&ctrl_cfg, DMA_SIZE_32);
92 channel_config_set_read_increment(&ctrl_cfg, true);
93 channel_config_set_write_increment(&ctrl_cfg, false);
94 /* Address wrapping to 2 words (2^3 = 8 bytes). We have 2 pointers, both
95 * are 4 bytes big. */
96 channel_config_set_ring(&ctrl_cfg, false, 3);
97 channel_config_set_chain_to(&ctrl_cfg, g_audio_chan);
98 channel_config_set_high_priority(&ctrl_cfg, true);
99
100 dma_channel_configure(
101 g_ctrl_chan, &ctrl_cfg, &dma_hw->ch[g_audio_chan].al3_read_addr_trig,
102 g_reload_addrs, 1, true
103 );
104}

106void audio_queue(ApuSample smp) {
107 g_bufs[g_curr_buf_idx][g_buf_pos++] = PWM_VALUE(smp);
108
109 if (g_buf_pos >= AUDIO_BUFFER_SIZE) {
110 unsigned next_buf_idx = (g_curr_buf_idx + 1) & (RING_SIZE - 1);
111
112 for (;;) {
113 uintptr_t dma_addr = dma_hw->ch[g_audio_chan].read_addr;
114 uintptr_t next_buf_start =
115 (uintptr_t) g_bufs[next_buf_idx] | 0x20000000;
116 uintptr_t next_buf_end = next_buf_start + sizeof(g_bufs[0]);
117
118 if (dma_addr < next_buf_start || dma_addr >= next_buf_end) {
119 break;
120 }
121
122 tight_loop_contents();
123 }
124
125 g_curr_buf_idx = next_buf_idx;
126 g_buf_pos = 0;
127 }
128}
