26void mapper4_update_prg(Disk *d) {
27 Mapper4 *m = &d->map4;
28 uint8_t *prg = d->prg;
29
30 uint8_t last = m->prg_bank_count - 1;
31 uint8_t second_last = (m->prg_bank_count >= 2) ? (last - 1) : 0;
32
33 uint8_t mask = m->prg_bank_count - 1;
34
35 uint8_t bank6 = m->bank_regs[6] & mask;
36 uint8_t bank7 = m->bank_regs[7] & mask;
37
38 if (m->prg_mode == 0) {
39 m->prg_banks[0] = &prg[bank6 << 13];
40 m->prg_banks[1] = &prg[bank7 << 13];
41 m->prg_banks[2] = &prg[second_last << 13];
42 m->prg_banks[3] = &prg[last << 13];
43 } else {
44 m->prg_banks[0] = &prg[second_last << 13];
45 m->prg_banks[1] = &prg[bank7 << 13];
46 m->prg_banks[2] = &prg[bank6 << 13];
47 m->prg_banks[3] = &prg[last << 13];
48 }
49}

51void mapper4_update_chr(Disk *d) {
52 Mapper4 *m = &d->map4;
53 uint8_t *chr = d->chr;
54
55 int count = d->chr_size >> 10;
56 if (count == 0) count = 1;
57
58 int mask = count - 1;
59
60 int r0 = (m->bank_regs[0] & ~1) & mask;
61 int r1 = (m->bank_regs[1] & ~1) & mask;
62
63 int r2 = m->bank_regs[2] & mask;
64 int r3 = m->bank_regs[3] & mask;
65 int r4 = m->bank_regs[4] & mask;
66 int r5 = m->bank_regs[5] & mask;
67
68 if (m->chr_mode == 0) {
69 m->chr_banks[0] = &chr[(r0 + 0) << 10];
70 m->chr_banks[1] = &chr[(r0 + 1) << 10];
71
72 m->chr_banks[2] = &chr[(r1 + 0) << 10];
73 m->chr_banks[3] = &chr[(r1 + 1) << 10];
74
75 m->chr_banks[4] = &chr[r2 << 10];
76 m->chr_banks[5] = &chr[r3 << 10];
77 m->chr_banks[6] = &chr[r4 << 10];
78 m->chr_banks[7] = &chr[r5 << 10];
79 } else {
80 m->chr_banks[0] = &chr[r2 << 10];
81 m->chr_banks[1] = &chr[r3 << 10];
82 m->chr_banks[2] = &chr[r4 << 10];
83 m->chr_banks[3] = &chr[r5 << 10];
84
85 m->chr_banks[4] = &chr[(r0 + 0) << 10];
86 m->chr_banks[5] = &chr[(r0 + 1) << 10];
87
88 m->chr_banks[6] = &chr[(r1 + 0) << 10];
89 m->chr_banks[7] = &chr[(r1 + 1) << 10];
90 }
91}

93void NOTFLASH_FN(mapper4_write)(Disk *d, uint16_t addr, uint8_t data) {
94 Mapper4 *m = &d->map4;
95
96 if (addr >= 0x6000 && addr < 0x8000) {
97 d->prg_ram[(addr - 0x6000) & (PRG_RAM_SIZE - 1)] = data;
98 return;
99 }
100
101 switch (addr & 0xE001) {
102
103 case 0x8000:
104 m->bank_select = data & 0x07;
105 m->prg_mode = (data >> 6) & 1;
106 m->chr_mode = (data >> 7) & 1;
107 break;
108 case 0x8001:
109 m->bank_regs[m->bank_select] = data;
110 mapper4_update_prg(d);
111 mapper4_update_chr(d);
112 break;
113
114 case 0xA000:
115 d->mirroring = (data & 1) ? MIRRORING_HORIZONTAL : MIRRORING_VERTICAL;
116 break;
117
118 case 0xC000:
119 m->irq_latch = data;
120
121 break;
122 case 0xC001:
123 m->irq_cnt = 0;
124 m->irq_reload_pending = true;
125
126 break;
127 case 0xE000:
128 m->irq_enabled = false;
129 cpu_irq_pulldown(&d->bus->cpu, IRQ_MMC3, false);
130
131 break;
132 case 0xE001:
133 m->irq_enabled = true;
134
135 break;
136 }
137}

139uint8_t NOTFLASH_FN(mapper4_read)(Disk *d, uint16_t addr) {
140 Mapper4 *m = &d->map4;
141
142 if (addr >= 0x8000) {
143 uint16_t rel = addr - 0x8000;
144 return m->prg_banks[(rel >> 13) & 3][rel & 0x1FFF];
145 }
146
147 if (addr >= 0x6000)
148 return d->prg_ram[(addr - 0x6000) & (PRG_RAM_SIZE - 1)];
149
150 return 0;
151}
