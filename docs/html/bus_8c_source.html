29static void read_disk(Bus *self) {
30 uint8_t pc_lsb = bus_read(self, 0xfffc);
31 uint8_t pc_msb = bus_read(self, 0xfffd);
32 self->cpu.PC = (uint16_t) ((pc_msb << 8) | pc_lsb);
33
34 self->ppu.mirroring = self->disk.mirroring;
35 self->ppu.chr = self->disk.chr;
36
37 if (self->ppu.mirroring == MIRRORING_VERTICAL) {
38 self->ppu.nt[0] = &self->ppu.vram[0x000];
39 self->ppu.nt[1] = &self->ppu.vram[0x400];
40 self->ppu.nt[2] = &self->ppu.vram[0x000];
41 self->ppu.nt[3] = &self->ppu.vram[0x400];
42 } else {
43 self->ppu.nt[0] = &self->ppu.vram[0x000];
44 self->ppu.nt[1] = &self->ppu.vram[0x000];
45 self->ppu.nt[2] = &self->ppu.vram[0x400];
46 self->ppu.nt[3] = &self->ppu.vram[0x400];
47 }
48}

56void bus_queue_add(BusQueue *self, const BusEvent *ev) {
57 if (self->count > BUS_EVENT_QUEUE_CAP) {
58 return;
59 } else if (self->count == 0) {
60 self->head = 0;
61 self->tail = 0;
62 } else {
63 self->tail = (self->tail + 1) % BUS_EVENT_QUEUE_CAP;
64 }
65
66 ++self->count;
67 memcpy(&self->events[self->tail], ev, sizeof(*ev));
68}
69
70BusEvent *bus_queue_pop(BusQueue *self, BusEvent *out) {
71 if (self->count == 0) {
72 return NULL;
73 }
74
75 BusEvent *event = NULL;
76
77 if (self->head == self->tail) {
78 event = &self->events[self->head];
79 self->head = 0;
80 self->tail = 0;
81 self->count = 0;
82 } else {
83 event = &self->events[self->head];
84 self->head = (self->head + 1) % BUS_EVENT_QUEUE_CAP;
85 --self->count;
86 }
87
88 if (out) {
89 memcpy(out, event, sizeof(*out));
90 }
91
92 return event;
93}

95bool bus_queue_peek(BusQueue *self, BusEvent *out) {
96 if (self->count == 0) {
97 return false;
98 }
99
100 memcpy(out, &self->events[self->head], sizeof(*out));
101
102 return true;
103}

105BusEvent *bus_queue_peek_ref(BusQueue *self) {
106 if (self->count == 0) {
107 return NULL;
108 }
109
110 return &self->events[self->head];

155static bool load_rom_mem(Bus *self, const uint8_t *data, unsigned size) {
156 /* To prevent trying to load an invalid .nes file when the currently
157 * connected disk is already unloaded, we will load the file into a
158 * temporary disk instance. */
159 static Disk disk;
160
161 if (!disk_load_mem(&disk, self, data, size)) {
162 return false;
163 }
164
165 if (self->disk_connected) {
166 disk_unload(&self->disk);
167 }
168
169 memcpy(&self->disk, &disk, sizeof(disk));
170 self->disk_connected = true;
171
172 bus_reset(self);
173 read_disk(self);
174
175 return true;
176}

178static bool NOTFLASH_FN(handle_event)(Bus *self, const BusEvent *ev) {
179 switch (ev->id) {
180 case BUS_EVENT_NONE:
181 break;
182 case BUS_EVENT_LOAD_ROM: {
183 if (!bus_load_disk(self, ev->load_rom.path)) {
184 BusEvent new_ev = {
185 .id = BUS_EVENT_SET_FWX_ERR,
186 .set_fwx_err = {.err = FWX_ERR_BAD_DATA},
187 };
188
189 bus_add_event(self, &new_ev);
190 }
191
192 return true;
193 }
194 case BUS_EVENT_SET_FWX_ERR:
195 fwx_set_error(&self->fwx, ev->set_fwx_err.err);
196
197 break;
198 case BUS_EVENT_RESET: {
199 /* TODO: handle failure of fwx_reset() */
200 fwx_reset(&self->fwx);
201
202#ifdef BUILD_RP2350
203 bus_load_disk(self, "/bios.nes");
204#else
205 bus_load_disk(self, "bios.nes");
206#endif
207
208 return true;
209 }
210 }
211
212 return false;
213}

215bool NOTFLASH_FN(bus_update)(Bus *self) {
216 while (self->ev_queue.count > 0) {
217 BusEvent *ev = bus_queue_peek_ref(&self->ev_queue);
218
219 if (ev && handle_event(self, ev)) {
220 return true;
221 }
222
223 bus_queue_pop(&self->ev_queue, NULL);
224 }
225
226 return false;

233void NOTFLASH_FN(bus_write)(Bus *self, uint16_t addr, uint8_t data) {
234 /* NES memory is organized in 4 KB pages */
235
236 switch (addr >> 12) {
237 case 0x0:
238 case 0x1:
239 /* Internal 2 KiB of RAM */
240
241 self->memory[addr & (0x0800 - 1)] = data;
242
243 break;
244 case 0x2:
245 case 0x3:
246 /* PPU registers, mirrored */
247
248 if (self->ppu.cycles < self->cpu.cycles * 3) {
249 ppu_run_until(&self->ppu, self->cpu.cycles * 3);
250 ppu_sync(&self->ppu);
251 }
252
253 addr = 0x2000 + (addr & 7);
254 ppu_write_reg(&self->ppu, addr, data);
255
256 break;
257 case 0x4:
258 if (addr == 0x4014) {
259 uint16_t high_addr = (uint16_t) (data << 8);
260
261 for (uint16_t i = 0; i < 256; ++i) {
262 ppu_oam_write(&self->ppu, bus_read(self, high_addr + i));
263 }
264
265 if (self->cpu.cycles & 1) {
266 self->cpu.cycles += 514;
267 } else {
268 self->cpu.cycles += 513;
269 }
270 }
271
272 if (addr == CONTROLLER_POLL_ADDR) {
273 joypad_write(&self->joypad1, data);
274 joypad_write(&self->joypad2, data);
275 } else if (addr >= 0x4000 && addr <= 0x4017) {
276 /* APU registers */
277 apu_run_until(&self->apu, self->cpu.cycles);
278 apu_write(&self->apu, addr, data);
279 } else if (addr >= 0x4018 && addr <= 0x401f) {
280 /* Disabled APU and I/O functionality, FWX */
281
282 fwx_write(&self->fwx, addr, data);
283 }
284
285 break;
286 default:
287 /* 0x4020..0xffff: unmapped/for cartridge use. */
288
289 self->disk.mapper_write(&self->disk, addr, data);
290 }
291}

293uint8_t NOTFLASH_FN(bus_read)(Bus *self, uint16_t addr) {
294 /* NES memory is organized in 4 KB pages. */
295
296 switch (addr >> 12) {
297 case 0x0:
298 case 0x1:
299 /* Internal 2 KiB of RAM */
300
301 return self->memory[addr & (0x0800 - 1)];
302 case 0x2:
303 case 0x3:
304 /* PPU registers, mirrored */
305
306 if (self->ppu.cycles < self->cpu.cycles * 3) {
307 ppu_run_until(&self->ppu, self->cpu.cycles * 3);
308 ppu_sync(&self->ppu);
309 }
310
311 return ppu_read_reg(&self->ppu, 0x2000 + (addr & (8 - 1)));
312 case 0x4:
313 if (addr == CONTROLLER1_ADDR) {
314 /* 0x4016: controller 1 */
315
316 return joypad_read(&self->joypad1);
317 } else if (addr == CONTROLLER2_ADDR) {
318 /* 0x4017: controller 2 */
319
320 return joypad_read(&self->joypad2);
321 // return controller_read(&self->controller2);
322 } else if (addr >= 0x4000 && addr <= 0x4017) {
323 /* APU registers */
324 apu_run_until(&self->apu, self->cpu.cycles);
325 return apu_read(&self->apu, addr);
326 } else if (addr >= 0x4018 && addr <= 0x401f) {
327 /* Disabled APU and I/O functionality, FWX */
328
329 return fwx_read(&self->fwx, addr);
330 }
331
332 break;
333 default:
334 /* 0x4020..0xffff: unmapped/for cartridge use. */
335
336 return self->disk.mapper_read(&self->disk, addr);
337 }
338
339 return 0;
340}
