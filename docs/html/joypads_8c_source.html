46void update_joypads(Joypad *joy1, Joypad *joy2) {
47 /* SYNC: in case PIO goes out of sync, we clear out the FIFO queue. */
48 while (!pio_sm_is_rx_fifo_empty(g_pio, g_sm)) {
49 pio_sm_get(g_pio, g_sm);
50 }
51
52 if (!pio_sm_is_tx_fifo_full(g_pio, g_sm)) {
53 pio_sm_put(g_pio, g_sm, 0);
54 }
55
56 /* invert because buttons are connected to GND */
57 uint32_t raw = pio_sm_get_blocking(g_pio, g_sm);
58 uint16_t data = (uint16_t) (raw & 0xffff);
59 data = (uint16_t) ~data;
60
61 joy1->state = 0x00;
62 joy2->state = 0x00;
63
64 for (unsigned i = 0; i < 8; ++i) {
65 uint8_t pair = (data >> (i * 2)) & 0x03;
66
67 if (pair & 1) {
68 joy1->state |= (1 << i);
69 }
70
71 if (pair & 2) {
72 joy2->state |= (1 << i);
73 }
74 }
75}
