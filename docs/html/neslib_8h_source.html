206//when display is enabled, vram access could only be done with this vram update system
207//the function sets a pointer to the update buffer that contains data and addresses
208//in a special format. It allows to write non-sequential bytes, as well as horizontal or
209//vertical nametable sequences.
210//buffer pointer could be changed during rendering, but it only takes effect on a new frame
211//number of transferred bytes is limited by vblank time
212//to disable updates, call this function with NULL pointer
213
214//the update data format:
215// MSB, LSB, byte for a non-sequential write
216// MSB|NT_UPD_HORZ, LSB, LEN, [bytes] for a horizontal sequence
217// MSB|NT_UPD_VERT, LSB, LEN, [bytes] for a vertical sequence
218// NT_UPD_EOF to mark end of the buffer
219
220//length of this data should be under 256 bytes
221
222void __fastcall__ set_vram_update(const unsigned char *buf);
