22#include "common.h"

24#define FWXSTAT *((volatile unsigned char *) 0x4018)
25#define FWXCTRL *((volatile unsigned char *) 0x4019)
26#define FWXDATA *((volatile unsigned char *) 0x401a)
27
28#define fwx_start() (FWXCTRL = FWX_CTRL_SS)
29#define fwx_stop() (FWXCTRL = 0)
30
31#define fwx_tx_done() (FWXSTAT & FWX_STAT_S2)
32#define fwx_rx_ready() (FWXSTAT & FWX_STAT_S1)
33
34#define fwx_write_u8(data) (FWXDATA = data, FWXCTRL |= FWX_CTRL_CLK)
35#define fwx_read_u8() (FWXCTRL |= FWX_CTRL_CLK, FWXDATA)
36
37#define fwx_wait_tx_done() while (!fwx_tx_done())
38#define fwx_wait_rx_ready() while (!fwx_rx_ready())
39
40#define fwx_last_error() (FWXSTAT & 0x0f)
41
42#endif /* FWX_PUBLIC_H */
