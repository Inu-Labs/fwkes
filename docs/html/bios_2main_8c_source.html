76static void fetch_filename(uint8_t idx) {
77 fwx_write_u8(FWX_CMD_SD_GET_ENTRY);
78 fwx_write_u8(idx);
79
80 type = fwx_read_u8();
81 is_dir = (type > 0);
82
83 filename_len = 0;
84
85 while (FWXSTAT & FWX_STAT_S1) {
86 char c = (char)fwx_read_u8();
87
88 if (c == 0) break;
89
90 if (filename_len < 29) {
91 filename[filename_len++] = c;
92 }
93 }
94
95 if (is_dir && filename_len < 29) {
96 filename[filename_len++] = '/';
97 }
98
99 filename[filename_len] = '\0';
100
101}

121static void draw_filename(uint8_t idx, bool loaded) {
122 static char idx_str[4];

126 idx_str[3] = 0;
127 print_at(0, y, idx_str);

130
131 for (j = 0; filename[j]; ++j) {
132 uint8_t ch = (uint8_t) filename[j];
133 vram_put(loaded ? ch + 0x70 : ch);

206static void move_cursor(int delta) {
207 old_cursor = cursor;
208 old_page = page;
209
210 cursor += delta;
211 wrap_cursor();
212
213 page = cursor_within_page();
214
215 if (page != old_page) {
216 ppu_off();
217 clear_page_area();
218 draw_page();
219 ppu_on_all();
220 }
221
222 draw_cursor();
223}

315static void handle_input(void) {
316 if (pad1_pressed & PAD_DOWN) {
317 move_cursor(+1);
318 } else if (pad1_pressed & PAD_UP) {
319 move_cursor(-1);
320 } else if (pad1_pressed & PAD_LEFT) {
321 move_cursor(-FILES_PER_PAGE);
322 } else if (pad1_pressed & PAD_RIGHT) {
323 move_cursor(+FILES_PER_PAGE);
324 } else if (pad1_pressed & PAD_START) {
325 load_rom();
326 } else if(pad1_pressed & PAD_A){
327 show_credits();
328 }
329}
