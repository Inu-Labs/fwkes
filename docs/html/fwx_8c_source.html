89static FwxError cmd_sd_get_entry(Fwx *self) {
90 unsigned idx = fwx_buf_get(&self->rx);
91
92 if (idx >= self->file_count) {
93 return FWX_ERR_BAD_IDX;
94 }
95
96 const char *path = self->files[idx];
97 unsigned len = (unsigned) strlen(path);
98
99 bool is_dir = (len > 0 && path[len - 1] == '/');
100
101 fwx_buf_put(&self->tx, is_dir ? 1 : 0);
102
103 if (is_dir && len > 0) {
104 len--;
105 }
106
107 for (unsigned i = 0; i < len; i++) {
108 fwx_buf_put(&self->tx, (uint8_t) path[i]);
109 }
110
111 fwx_buf_put(&self->tx, 0);
112
113 return FWX_ERR_OK;
114}

141static void handle_clk(Fwx *self) {
142 if (!(self->regs.STAT & FWX_STAT_S1)) {
143 fwx_buf_put(&self->rx, self->regs.DATA);
144
145 trace_fwx(
146 TRACE_RX_DATA, "rx.data[%u] = 0x%02x (%c)\n", self->rx.pos - 1,
147 self->regs.DATA, (char) self->regs.DATA
148 );
149
150 if (self->rx.size == 1) {
151 FwxCmdId cmd = self->rx.data[0];
152
153 if (cmd >= sizeof(g_cmd_table) / sizeof(g_cmd_table[0])) {
154 fwx_buf_reset(&self->rx);
155 fwx_buf_rewind(&self->tx);
156
157 return;
158 }
159
160 self->curr_cmd = cmd;
161 self->rx.expected_size = g_cmd_size[cmd];
162 }
163
164 if (self->rx.size == self->rx.expected_size) {
165 self->regs.STAT |= FWX_STAT_S2;
166 self->rx.pos = 1;
167
168 fwx_execute_cmd(self, self->curr_cmd);
169
170 fwx_buf_reset(&self->rx);
171 fwx_buf_rewind(&self->tx);
172
173 if (self->tx.size > 0) {
174 self->regs.STAT |= FWX_STAT_S1;
175 }
176 }
177 } else {
178 if (self->tx.pos < self->tx.size) {
179 uint8_t data = fwx_buf_get(&self->tx);
180 self->regs.DATA = data;
181
182 trace_fwx(
183 TRACE_TX_DATA, "tx.data[%u] = 0x%02x (%c)\n", self->tx.pos - 1,
184 self->regs.DATA, (char) self->regs.DATA
185 );
186 }
187
188 if (self->tx.pos >= self->tx.size) {
189 self->regs.STAT &= ~FWX_STAT_S1;
190 fwx_buf_reset(&self->tx);
191 }
192 }
193
194 self->regs.CTRL &= ~FWX_CTRL_CLK;
195}

265static FsError load_rom(Fwx *self, const char *path) {
266 FileType type;
267
268 FsError err = fs_type(path, &type);
269 PROPAGATE_IF_FS_ERR(err);
270
271 if (type == FILE_FOLDER) {
272 err = fs_cwd(self->fs, path);
273 PROPAGATE_IF_FS_ERR(err);
274
275 err = open_cwd(self);
276 PROPAGATE_IF_FS_ERR(err);
277
278 err = fwx_update_file_list(self);
279 PROPAGATE_IF_FS_ERR(err);
280 } else {
281 BusEvent ev = {
282 .id = BUS_EVENT_LOAD_ROM,
283 .load_rom = {.path = path},
284 };
285
286 bus_add_event(self->bus, &ev);
287 }
288
289 return FS_ERR_OK;

457FsError fwx_update_file_list(Fwx *self) {
458 self->file_count = 0;
459
460 for (int i = 0; i < 256; ++i) {
461 memset(self->files[i], 0, 256);
462 }
463
464 dir_rewind(&self->cwd);
465
466 DirEntry de;
467
468#ifdef BUILD_RP2350
469 /* FatFs' readdir does not return '..' in contrast to Linux' readdir. */
470 strncpy(self->files[0], "../", FILENAME_MAX_SIZE);
471 ++self->file_count;
472#endif
473
474 while (self->file_count < 255) {
475 FsError err = dir_read(&self->cwd, &de);
476
477 if (err == FS_ERR_DIR_END) {
478 break;
479 } else if (err != FS_ERR_OK) {
480 return err;
481 }
482
483 if (strcmp(de.name, ".") == 0) {
484 continue;
485 }
486
487#if BUILD_RP2350
488 if (strcmp(de.name, ".sd_checker") != 0) {
489#endif
490 FileType type;
491 if (fs_type(de.name, &type) == FS_ERR_OK && type == FILE_FOLDER) {
492 snprintf(
493 self->files[self->file_count], FILENAME_MAX_SIZE, "%s/",
494 de.name
495 );
496 } else {
497 strncpy(
498 self->files[self->file_count], de.name, FILENAME_MAX_SIZE
499 );
500 }
501
502 ++self->file_count;
503#if BUILD_RP2350
504 }
505#endif
506 }
507
508 qsort(self->files, self->file_count, 256, qsort_strcmp);
509
510 return FS_ERR_OK;
511}

584void fwx_write_ctrl(Fwx *self, uint8_t value) {
585 uint8_t prev = self->regs.CTRL;
586 self->regs.CTRL = value;
587
588 /* START condition */
589 if (!(prev & FWX_CTRL_SS) && (value & FWX_CTRL_SS)) {
590 trace_fwx(TRACE_START_COND, "");
591 reset_state(self);
592
593 return;
594 }
595
596 /* STOP condition */
597 if ((prev & FWX_CTRL_SS) && !(value & FWX_CTRL_SS)) {
598 trace_fwx(TRACE_STOP_COND, "");
599 self->regs.STAT &= ~(FWX_STAT_S0 | FWX_STAT_S1 | FWX_STAT_S2);
600 clear_error(self);
601
602 return;
603 }
604
605 /* CLK edge clock */
606 if ((self->regs.CTRL & FWX_CTRL_SS) && (value & FWX_CTRL_CLK)) {
607 trace_fwx(TRACE_CLK_EDGE, "");
608 handle_clk(self);
609 }
