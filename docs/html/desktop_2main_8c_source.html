240void emu_handle_events(Emulator *self) {
241 SDL_Event ev;
242 while (SDL_PollEvent(&ev)) {
243 if (ev.type == SDL_EVENT_QUIT) {
244 self->quit = true;
245 } else if (ev.type == SDL_EVENT_KEY_DOWN &&
246 ev.key.scancode == SDL_SCANCODE_ESCAPE) {
247 self->quit = true;
248 } else if (ev.type == SDL_EVENT_KEY_DOWN &&
249 ev.key.scancode == SDL_SCANCODE_G) {
250 self->draw_grid = !self->draw_grid;
251 } else if (ev.type == SDL_EVENT_KEY_DOWN &&
252 ev.key.scancode == SDL_SCANCODE_R) {
253 BusEvent new_ev = {
254 .id = BUS_EVENT_RESET,
255 };
256
257 bus_add_event(&self->bus, &new_ev);
258 }
259 }
260
261 const Uint8 *keyboard_state = (const Uint8 *) SDL_GetKeyboardState(NULL);
262 update_joypad_state(&self->bus.joypad1, keyboard_state);
263 update_joypad_state(&self->bus.joypad2, keyboard_state);
264}

266static void run_frame(Emulator *self) {
267 Cpu *cpu = &self->bus.cpu;
268 Ppu *ppu = &self->bus.ppu;
269 Apu *apu = &self->bus.apu;
270
271 while (!ppu->frame_done) {
272 /* CPU will run until some significant event (vblank/nmi or end of
273 * frame)
274 */
275
276 CycleCounter curr_ppu_abs_dot =
277 ppu->scanline * DOTS_PER_SCANLINE + ppu->dot;
278 CycleCounter target_ppu_abs_dot;
279
280 if (self->bus.disk.mapper_hsync && ppu->scanline < 240) {
281 if (ppu->dot < 260) {
282 /* Target the current scanline's IRQ dot */
283 target_ppu_abs_dot = ppu->scanline * DOTS_PER_SCANLINE + 260;
284 } else {
285 /* We passed it, target the NEXT scanline's IRQ dot */
286 target_ppu_abs_dot =
287 (ppu->scanline + 1) * DOTS_PER_SCANLINE + 260;
288 }
289 } else if (ppu->scanline < 241) {
290 /* next event is vblank */
291 target_ppu_abs_dot = 241 * DOTS_PER_SCANLINE + 1;
292 } else {
293 /* next event is end of frame */
294 target_ppu_abs_dot = 262 * DOTS_PER_SCANLINE;
295 }
296
297 CycleCounter cycles_to_event =
298 (target_ppu_abs_dot - curr_ppu_abs_dot) / 3;
299 CycleCounter event_deadline = cpu->cycles + cycles_to_event;
300 CycleCounter cycles_to_run = cpu->cycles + CYCLES_PER_SCANLINE;
301
302 /* significant event has higher priority */
303 if (event_deadline < cycles_to_run) {
304 cycles_to_run = event_deadline;
305 }
306
307 /* prevent infinite loop */
308 if (cycles_to_run <= cpu->cycles) {
309 cycles_to_run = cpu->cycles + 1;
310 }
311
312 cpu_run_until(cpu, cycles_to_run);
313 ppu_run_until(ppu, cpu->cycles * 3);
314 apu_run_until(apu, cpu->cycles);
315
316 // bus_hsync(&self->bus);
317 }
318
319 ppu->frame_done = false;
320}

322void emu_run(Emulator *self) {
323 while (!self->quit) {
324 uint64_t start_time = SDL_GetTicksNS();
325
326 emu_handle_events(self);
327
328 bus_update(&self->bus);
329 run_frame(self);
330
331 SDL_SetRenderDrawColor(self->renderer, 0x00, 0x00, 0x00, 0xff);
332 SDL_RenderClear(self->renderer);
333
334 if (self->frame_done) {
335 emu_render(self);
336 }
337
338 SDL_RenderPresent(self->renderer);
339
340 uint64_t end_time = start_time + FRAME_TIME;
341 uint64_t now = SDL_GetTicksNS();
342
343 if (now < end_time) {
344 SDL_DelayPrecise(end_time - now);
345 }
346 }
347}
