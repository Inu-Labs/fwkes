81void vlog_msg_(LogCategory cat, LogType type, const char *fmt, va_list vargs) {
82 if ((g_cat_filter & cat) || (g_type_filter & type)) {
83 return;
84 }
85
86 if (!g_stream) {
87 g_stream = stdout;
88 }
89
90 unsigned cat_idx = cat_to_idx(cat);
91 unsigned type_idx = type_to_idx(type);
92 const char *cat_str = g_cat_str[cat_idx];
93 const char *type_str = g_type_str[type_idx];
94
95#if _WIN32
96 if (_isatty(_fileno(g_stream))) {
97#elif __unix__
98 if (isatty(fileno(g_stream))) {
99#else
100 if(0) {
101#endif
102 fputs(g_cat_color_mod[cat_idx], g_stream);
103 fputs(g_type_color_mod[type_idx], g_stream);
104 fputs(cat_str, g_stream);
105 fputs(type_str, g_stream);
106 fputs(": ", g_stream);
107 vfprintf(g_stream, fmt, vargs);
108 fputs("\033[0m\n", g_stream);
109 } else {
110 fputs(cat_str, g_stream);
111 fputs(type_str, g_stream);
112 fputs(": ", g_stream);
113 vfprintf(g_stream, fmt, vargs);
114 fputc('\n', g_stream);
115 }
116}
